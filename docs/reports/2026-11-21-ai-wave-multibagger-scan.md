## 偏差標註

1. 海外只有名冊上的公司——名冊是我們照題材挑的，「名冊海外 ≥10 倍 63／182」不是任何市場的基準率；名冊外的海外十倍股全部漏掉。
2. 台股宇宙是 2023-12 還在月營收頁上的公司；之後下市的有、之前下市的沒有。2022-10 之後才掛牌的從第一根算。
3. 月收盤：盤中高低與月內的急漲急跌看不到；「達 N 倍的月」最多晚一個月。
4. 清單沒有分「是不是 AI」：生技、營建、太陽能等與這一波無關的也在裡面。層的歸屬要另外標，節點標籤不夠用。
5. 2026-11 是未收完的月，而且這個月跌幅很大（例：清單內現價對高點中位 0.40）。

"""AI 這一波的倍數股機械掃描（2022-10 起；使用者 2026-11-21 問「AI 這波的十倍股有哪些」）。
宇宙＝全台股上市櫃（月營收頁 2023-12 的代號）＋名冊 research_ticker。資料＝yfinance 月線（還原權息）。
零 LLM；只寫 scratchpad。對外請求不帶任何個人資訊。
量：2022-10 之後「先低後高」的最大倍數（月收盤）、低點月、第一次達 2／5／10 倍的月、高點月、現價對高點。"""
import json
import re
import sys
import time
from pathlib import Path

import yfinance as yf

HERE = Path(__file__).parent
DRY = HERE.parent / "dryrun"
REPO = Path(sys.argv[1])
START = "2022-09-01"
FIRST = "2022-10"


def log(*a):
    print(*a, flush=True)


# ---------- 宇宙 ----------
names, tickers = {}, []
for market, suffix in (("twse", ".TW"), ("tpex", ".TWO")):
    for r in json.loads((DRY / "rev" / f"rev_{market}_2023-12.json").read_text(encoding="utf-8")):
        code = r.get("company_code")
        if code and re.fullmatch(r"\d{4}", code):
            t = code + suffix
            if t not in names:
                names[t] = r.get("company_name") or code
                tickers.append(t)
tw_n = len(tickers)
reg = json.loads((REPO / "config" / "company_identity.json").read_text(encoding="utf-8"))["companies"]
for c in reg:
    t = c.get("research_ticker")
    if t and t not in names:
        names[t] = c.get("display_name") or t
        tickers.append(t)
log("宇宙：台股", tw_n, "名冊另加", len(tickers) - tw_n, "合計", len(tickers))

# 台股節點標籤（櫃買平台，今天的版本）
labels = {}
for f in sorted((DRY / "tpex").glob("*.html")):
    if f.stem == "index":
        continue
    t = f.read_text(encoding="utf-8", errors="replace")
    chain = re.sub(r"\s+", " ", (re.findall(r"<title>(.*?)</title>", t, flags=re.S) or [""])[0])
    chain = chain.replace("產業價值鏈資訊平台 >", "").replace("產業鏈簡介", "").strip()
    for part in re.split(r'<div id="companyList_', t)[1:]:
        m = re.match(r'([A-Za-z0-9]+)" title="([^"]*)"', part)
        if m:
            for code in set(re.findall(r"stk_code=([0-9A-Za-z]+)", part)):
                labels.setdefault(code, []).append(f"{chain}／{m.group(2)}")

# ---------- 月線 ----------
cache = HERE / "px.json"
px = json.loads(cache.read_text(encoding="utf-8")) if cache.exists() else {}
errors = {}


def pull(chunk, pause):
    try:
        yf.shared._ERRORS.clear()
    except Exception:  # noqa: BLE001
        pass
    try:
        df = yf.download(tickers=chunk, start=START, interval="1mo", auto_adjust=True,
                         progress=False, threads=True, group_by="ticker")
    except Exception as exc:  # noqa: BLE001
        for t in chunk:
            errors[t] = f"download: {exc}"
        time.sleep(pause * 3)
        return
    for t in chunk:
        try:
            s = (df[t]["Close"] if len(chunk) > 1 else df["Close"]).dropna()
            if hasattr(s, "columns"):
                s = s.iloc[:, 0]
        except Exception:  # noqa: BLE001
            s = None
        if s is not None and len(s) > 0:
            px[t] = [[d.strftime("%Y-%m"), float(v)] for d, v in s.items()]
            errors.pop(t, None)
        else:
            errors[t] = str(getattr(yf.shared, "_ERRORS", {}).get(t, "empty"))
    time.sleep(pause)


todo = [t for t in tickers if t not in px]
for i in range(0, len(todo), 80):
    pull(todo[i:i + 80], 1.5)
    if (i // 80) % 5 == 0:
        log("第一輪", i + 80, "/", len(todo), "已有", len(px))
        cache.write_text(json.dumps(px), encoding="utf-8")
miss = [t for t in tickers if t not in px]
log("第一輪後缺", len(miss))
for i in range(0, len(miss), 20):
    pull(miss[i:i + 20], 5)
cache.write_text(json.dumps(px), encoding="utf-8")
miss = [t for t in tickers if t not in px]
limited = [t for t in miss if re.search(r"rate|too many|429", errors.get(t, ""), flags=re.I)]
log("最後缺", len(miss), "其中錯誤訊息像限流的", len(limited))

# ---------- 量 ----------
rows = []
for t in tickers:
    s = [(m, v) for m, v in px.get(t, []) if m >= FIRST and v and v > 0]
    if len(s) < 6:
        continue
    best = (1.0, 0, 0)
    lo = 0
    for j in range(1, len(s)):
        if s[j - 1][1] < s[lo][1]:
            lo = j - 1
        mult = s[j][1] / s[lo][1]
        if mult > best[0]:
            best = (mult, lo, j)
    mult, lo, hi = best
    base = s[lo][1]

    def first(k):
        for m, v in s[lo + 1:]:
            if v >= k * base:
                return m
        return None

    code = t.split(".")[0]
    rows.append({"ticker": t, "name": names.get(t, ""), "mult": round(mult, 2), "low": s[lo][0], "t2": first(2),
                 "t5": first(5), "t10": first(10), "peak": s[hi][0], "now_vs_peak": round(s[-1][1] / s[hi][1], 2),
                 "first_bar": s[0][0], "last_bar": s[-1][0], "labels": labels.get(code, [])[:2],
                 "tw": t.endswith((".TW", ".TWO"))})
(HERE / "rows.json").write_text(json.dumps(rows, ensure_ascii=False), encoding="utf-8")
tw = [r for r in rows if r["tw"]]
log("\n== 計數（2022-10 起先低後高的最大倍數，月收盤）")
for lab, grp in (("台股", tw), ("名冊（非台股）", [r for r in rows if not r["tw"]])):
    log(lab, "有價", len(grp), "≥2倍", sum(r["mult"] >= 2 for r in grp), "≥5倍", sum(r["mult"] >= 5 for r in grp),
        "≥10倍", sum(r["mult"] >= 10 for r in grp))
log("\n== ≥10 倍，依第一次達 2 倍的月排序：代號｜名稱｜倍數｜低點月｜2倍月｜5倍月｜10倍月｜高點月｜現價/高點｜節點")
for r in sorted((r for r in rows if r["mult"] >= 10), key=lambda r: (r["t2"] or "9999", r["ticker"])):
    log(f"{r['ticker']}｜{r['name']}｜{r['mult']}｜{r['low']}｜{r['t2']}｜{r['t5']}｜{r['t10']}｜{r['peak']}｜{r['now_vs_peak']}｜{'；'.join(r['labels'])}")
log("SCAN_DONE")

"""把 AI 這一波倍數股掃描組成報告（輸出表與腳本逐字）。bytes 寫、LF 換行。"""
import json
import sys
from collections import Counter
from pathlib import Path
from statistics import median

HERE = Path(__file__).parent
OUT = Path(sys.argv[1])
rows = json.loads((HERE / "rows.json").read_text(encoding="utf-8"))
for r in rows:
    if not r["tw"]:
        r["labels"] = []  # 節點標籤只對台股代號有意義（日股數字代號會撞號）


def months(a, b):
    ya, ma = map(int, a.split("-"))
    yb, mb = map(int, b.split("-"))
    return (yb - ya) * 12 + (mb - ma)


tw = [r for r in rows if r["tw"]]
ov = [r for r in rows if not r["tw"]]
big = sorted((r for r in rows if r["mult"] >= 10), key=lambda r: (r["t2"] or "9999", r["ticker"]))
gap = [months(r["t2"], r["t10"]) for r in big if r["t2"] and r["t10"]]
